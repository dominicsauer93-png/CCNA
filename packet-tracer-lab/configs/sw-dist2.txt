! D2 (SW-DIST2) — distribution/core L3 switch #2 (mirror of D1, swapped active roles)
! Routed ports: Gi1/0/1 = R1, Gi1/0/2 = R2 (backup WAN, static only)
! Po1 = peer link to D1 (Gi1/0/3-4), Po10 = uplink to ACC2 (Gi1/0/5-6)
! Paste in global config mode (enable -> configure terminal)

hostname SW-DIST2
ip routing
no ip domain-lookup
ip domain-name lab.local

enable secret Lab-Enable1!
service password-encryption

username admin privilege 15 password Lab-Admin1!

banner motd ^C
***********************************************************
* SW-DIST2 - CCNA-LAB distribution switch #2.             *
* Authorized lab use only. All access logged.             *
***********************************************************
^C

crypto key generate rsa general-keys modulus 2048
ip ssh version 2

ip access-list standard MGMT-ACCESS
 permit 192.168.99.0 0.0.0.255
 deny   any

! ---- Extended ACL: block VOICE VLAN from SSH/Telnet into SERVERS VLAN ----
ip access-list extended VOICE-RESTRICT
 deny tcp 192.168.20.0 0.0.0.255 192.168.30.0 0.0.0.255 eq 22
 deny tcp 192.168.20.0 0.0.0.255 192.168.30.0 0.0.0.255 eq 23
 permit ip any any

line console 0
 login local
 exec-timeout 10 0
 logging synchronous
line vty 0 15
 transport input ssh
 login local
 access-class MGMT-ACCESS in
 exec-timeout 10 0
 logging synchronous

service timestamps log datetime msec
logging buffered 16384
logging host 192.168.30.10

clock timezone AEST 10
ntp server 10.0.13.1

! If your Packet Tracer version rejects SNMPv3 syntax below, fall back to:
!   snmp-server community LabRO1! RO
snmp-server group SNMP-ADMINS v3 priv
snmp-server user snmpadmin SNMP-ADMINS v3 auth sha Lab-SnmpAuth1! priv aes 128 Lab-SnmpPriv1!
snmp-server host 192.168.30.10 version 3 priv snmpadmin

cdp run
lldp run

! ---- VLANs ----
vlan 10
 name DATA
vlan 20
 name VOICE
vlan 30
 name SERVERS
vlan 50
 name WIRELESS
vlan 99
 name MGMT
vlan 999
 name NATIVE_UNUSED

! ---- Routed port to R1 (OSPF point-to-point) ----
interface GigabitEthernet1/0/1
 description Routed link to R1-EDGE
 no switchport
 ip address 10.0.13.2 255.255.255.252
 ip ospf network point-to-point
 no shutdown

! ---- Routed port to R2-BRANCH (backup WAN, static routing only - intentionally NOT in OSPF) ----
interface GigabitEthernet1/0/2
 description Routed link to R2-BRANCH (backup WAN, static-only)
 no switchport
 ip address 172.16.13.1 255.255.255.252
 no shutdown

! ---- Po1: peer EtherChannel trunk to D1 ----
interface range GigabitEthernet1/0/3 - 4
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate
 channel-group 1 mode active
 no shutdown
interface Port-channel1
 description Peer link to SW-DIST1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate

! ---- Po10: EtherChannel trunk uplink to ACC2 ----
interface range GigabitEthernet1/0/5 - 6
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate
 channel-group 10 mode active
 no shutdown
interface Port-channel10
 description Uplink to SW-ACC2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate
 ip dhcp snooping trust
 ip arp inspection trust

interface Loopback0
 description OSPF Router ID
 ip address 3.3.3.3 255.255.255.255

! ---- SVIs: D2 is the primary HSRP/STP-root/OSPF-DR for VLANs 20, 50 ----
interface Vlan10
 description DATA
 ip address 192.168.10.3 255.255.255.0
 ip helper-address 192.168.30.10
 ip ospf priority 0
 standby 10 ip 192.168.10.1
 standby 10 priority 90
 standby 10 preempt
 standby 10 authentication CcnaLab1
 no shutdown

interface Vlan20
 description VOICE
 ip address 192.168.20.3 255.255.255.0
 ip helper-address 192.168.30.10
 ip ospf priority 200
 standby 20 ip 192.168.20.1
 standby 20 priority 110
 standby 20 preempt
 standby 20 authentication CcnaLab1
 no shutdown

interface Vlan30
 description SERVERS
 ip address 192.168.30.3 255.255.255.0
 ip ospf priority 0
 standby 30 ip 192.168.30.1
 standby 30 priority 90
 standby 30 preempt
 standby 30 authentication CcnaLab1
 no shutdown

interface Vlan50
 description WIRELESS
 ip address 192.168.50.3 255.255.255.0
 ip helper-address 192.168.30.10
 ip ospf priority 200
 standby 50 ip 192.168.50.1
 standby 50 priority 110
 standby 50 preempt
 standby 50 authentication CcnaLab1
 no shutdown

interface Vlan99
 description MGMT
 ip address 192.168.99.3 255.255.255.0
 ip ospf priority 0
 standby 99 ip 192.168.99.1
 standby 99 priority 90
 standby 99 preempt
 standby 99 authentication CcnaLab1
 no shutdown

! ---- Apply the VOICE-RESTRICT ACL on the VLAN 20 SVI ----
interface Vlan20
 ip access-group VOICE-RESTRICT in

! ---- OSPF ----
router ospf 1
 router-id 3.3.3.3
 network 3.3.3.3 0.0.0.0 area 0
 network 10.0.13.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 network 192.168.30.0 0.0.0.255 area 0
 network 192.168.50.0 0.0.0.255 area 0
 network 192.168.99.0 0.0.0.255 area 0

! NOTE: D2 deliberately has NO static route to 192.168.40.0/24 (the branch LAN).
! Only D1 carries it (see sw-dist1.txt), so the backup WAN path is exercised
! from R2's side only (floating static) — see README §7 "Bonus" for how to
! extend this with redistribution if you want full backup-path reachability.

! ---- Spanning tree: D2 is root for 20/50, secondary for 10/30/99 ----
spanning-tree mode rapid-pvst
spanning-tree vlan 20,50 root primary
spanning-tree vlan 10,30,99 root secondary
spanning-tree portfast bpduguard default
spanning-tree loopguard default

! ---- Layer 2 threat mitigation on client-facing VLANs ----
ip dhcp snooping
ip dhcp snooping vlan 10,20,50,99
no ip dhcp snooping information option
ip arp inspection vlan 10,20,50,99

end
copy running-config startup-config

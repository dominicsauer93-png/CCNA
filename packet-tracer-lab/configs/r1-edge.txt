! R1-EDGE — Internet edge / NAT router
! Interfaces: Gi0/0 = ISP0 (outside), Gi0/1 = D1, Gi0/2 = D2, Lo0 = OSPF RID
! Paste in global config mode (enable -> configure terminal)

hostname R1-EDGE
no ip domain-lookup
ip domain-name lab.local

enable secret Lab-Enable1!
service password-encryption

username admin privilege 15 password Lab-Admin1!

banner motd ^C
***********************************************************
* R1-EDGE - CCNA-LAB Internet edge / NAT router.          *
* Authorized lab use only. All access logged.             *
***********************************************************
^C

crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3

ip access-list standard MGMT-ACCESS
 permit 192.168.99.0 0.0.0.255
 deny   any

! ---- Inbound ACL for the outside (ISP-facing) interface, and NAT permit list ----
! Defined here so they exist before being referenced under the interfaces/NAT below.
ip access-list extended OUTSIDE-IN
 permit icmp any any echo-reply
 permit icmp any any ttl-exceeded
 permit icmp any any unreachable
 permit tcp any host 203.0.113.5 eq 80
 permit udp any host 203.0.113.5 eq 69
 permit tcp any any established
 deny   ip any any

access-list 1 permit 192.168.0.0 0.0.255.255

line console 0
 login local
 exec-timeout 10 0
 logging synchronous
line vty 0 15
 transport input ssh
 login local
 access-class MGMT-ACCESS in
 exec-timeout 10 0
 logging synchronous

service timestamps log datetime msec localtime show-timezone
logging buffered 16384 informational
logging host 192.168.30.10
logging trap informational

clock timezone AEST 10
ntp server 203.0.113.1 prefer

! If your Packet Tracer version rejects SNMPv3 syntax below, fall back to:
!   snmp-server community LabRO1! RO
snmp-server group SNMP-ADMINS v3 priv
snmp-server user snmpadmin SNMP-ADMINS v3 auth sha Lab-SnmpAuth1! priv aes 128 Lab-SnmpPriv1!
snmp-server host 192.168.30.10 version 3 priv snmpadmin

no ip http server
no ip http secure-server

interface Loopback0
 description OSPF Router ID
 ip address 1.1.1.1 255.255.255.255

interface GigabitEthernet0/0
 description Outside - link to ISP0
 ip address 203.0.113.2 255.255.255.248
 ip nat outside
 ip access-group OUTSIDE-IN in
 no shutdown

interface GigabitEthernet0/1
 description Inside - link to D1
 ip address 10.0.12.1 255.255.255.252
 ip ospf network point-to-point
 ip nat inside
 no shutdown

interface GigabitEthernet0/2
 description Inside - link to D2
 ip address 10.0.13.1 255.255.255.252
 ip ospf network point-to-point
 ip nat inside
 no shutdown

! ---- Static default route toward the Internet ----
ip route 0.0.0.0 0.0.0.0 203.0.113.1

! ---- OSPF: peer with D1/D2, inject the default route learned above ----
router ospf 1
 router-id 1.1.1.1
 network 1.1.1.1 0.0.0.0 area 0
 network 10.0.12.0 0.0.0.3 area 0
 network 10.0.13.0 0.0.0.3 area 0
 default-information originate

! ---- NAT: PAT overload for all campus/branch traffic, static NAT for Server1 ----
ip nat inside source list 1 interface GigabitEthernet0/0 overload
ip nat inside source static 192.168.30.10 203.0.113.5

end
copy running-config startup-config

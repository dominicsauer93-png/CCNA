! D1 (SW-DIST1) — distribution/core L3 switch #1
! Routed ports: Gi1/0/1 = R1, Gi1/0/2 = R2 (primary WAN, static only)
! Po1 = peer link to D2 (Gi1/0/3-4), Po10 = uplink to ACC1 (Gi1/0/5-6)
! Paste in global config mode (enable -> configure terminal)

hostname SW-DIST1
ip routing
no ip domain-lookup
ip domain-name lab.local

enable secret Lab-Enable1!
service password-encryption

username admin privilege 15 password Lab-Admin1!

banner motd ^C
***********************************************************
* SW-DIST1 - CCNA-LAB distribution switch #1.             *
* Authorized lab use only. All access logged.             *
***********************************************************
^C

crypto key generate rsa general-keys modulus 2048
ip ssh version 2

ip access-list standard MGMT-ACCESS
 permit 192.168.99.0 0.0.0.255
 deny   any

line console 0
 login local
 exec-timeout 10 0
 logging synchronous
line vty 0 15
 transport input ssh
 login local
 access-class MGMT-ACCESS in
 exec-timeout 10 0
 logging synchronous

service timestamps log datetime msec localtime show-timezone
logging buffered 16384 informational
logging host 192.168.30.10
logging trap informational

clock timezone EST -5
ntp server 10.0.12.1 prefer

! If your Packet Tracer version rejects SNMPv3 syntax below, fall back to:
!   snmp-server community LabRO1! RO
snmp-server group SNMP-ADMINS v3 priv
snmp-server user snmpadmin SNMP-ADMINS v3 auth sha Lab-SnmpAuth1! priv aes 128 Lab-SnmpPriv1!
snmp-server host 192.168.30.10 version 3 priv snmpadmin

no ip http server
no ip http secure-server
cdp run
lldp run

! ---- VLANs ----
vlan 10
 name DATA
vlan 20
 name VOICE
vlan 30
 name SERVERS
vlan 50
 name WIRELESS
vlan 99
 name MGMT
vlan 999
 name NATIVE_UNUSED

! ---- Routed port to R1 (OSPF point-to-point) ----
interface GigabitEthernet1/0/1
 description Routed link to R1-EDGE
 no switchport
 ip address 10.0.12.2 255.255.255.252
 ip ospf network point-to-point
 no shutdown

! ---- Routed port to R2-BRANCH (primary WAN, static routing only - intentionally NOT in OSPF) ----
interface GigabitEthernet1/0/2
 description Routed link to R2-BRANCH (primary WAN, static-only)
 no switchport
 ip address 172.16.12.1 255.255.255.252
 no shutdown

! ---- Po1: peer EtherChannel trunk to D2 ----
interface range GigabitEthernet1/0/3 - 4
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate
 channel-group 1 mode active
 no shutdown
interface Port-channel1
 description Peer link to SW-DIST2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate

! ---- Po10: EtherChannel trunk uplink to ACC1 ----
interface range GigabitEthernet1/0/5 - 6
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate
 channel-group 10 mode active
 no shutdown
interface Port-channel10
 description Uplink to SW-ACC1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate
 ip dhcp snooping trust
 ip arp inspection trust

interface Loopback0
 description OSPF Router ID
 ip address 2.2.2.2 255.255.255.255

! ---- SVIs: D1 is the primary HSRP/STP-root/OSPF-DR for VLANs 10, 30, 99 ----
interface Vlan10
 description DATA
 ip address 192.168.10.2 255.255.255.0
 ip helper-address 192.168.30.10
 ip ospf priority 200
 standby 10 ip 192.168.10.1
 standby 10 priority 110
 standby 10 preempt
 standby 10 authentication CcnaLab1
 no shutdown

interface Vlan20
 description VOICE
 ip address 192.168.20.2 255.255.255.0
 ip helper-address 192.168.30.10
 ip ospf priority 0
 standby 20 ip 192.168.20.1
 standby 20 priority 90
 standby 20 preempt
 standby 20 authentication CcnaLab1
 no shutdown

interface Vlan30
 description SERVERS
 ip address 192.168.30.2 255.255.255.0
 ip ospf priority 200
 standby 30 ip 192.168.30.1
 standby 30 priority 110
 standby 30 preempt
 standby 30 authentication CcnaLab1
 no shutdown

interface Vlan50
 description WIRELESS
 ip address 192.168.50.2 255.255.255.0
 ip helper-address 192.168.30.10
 ip ospf priority 0
 standby 50 ip 192.168.50.1
 standby 50 priority 90
 standby 50 preempt
 standby 50 authentication CcnaLab1
 no shutdown

interface Vlan99
 description MGMT
 ip address 192.168.99.2 255.255.255.0
 ip ospf priority 200
 standby 99 ip 192.168.99.1
 standby 99 priority 110
 standby 99 preempt
 standby 99 authentication CcnaLab1
 no shutdown

! ---- OSPF ----
router ospf 1
 router-id 2.2.2.2
 network 2.2.2.2 0.0.0.0 area 0
 network 10.0.12.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 network 192.168.30.0 0.0.0.255 area 0
 network 192.168.50.0 0.0.0.255 area 0
 network 192.168.99.0 0.0.0.255 area 0

! ---- Static route to the branch LAN via the primary WAN link (3.3.b network route) ----
ip route 192.168.40.0 255.255.255.0 172.16.12.2

! ---- Spanning tree: D1 is root for 10/30/99, secondary for 20/50 ----
spanning-tree mode rapid-pvst
spanning-tree vlan 10,30,99 root primary
spanning-tree vlan 20,50 root secondary
spanning-tree portfast bpduguard default
spanning-tree loopguard default

! ---- Layer 2 threat mitigation on client-facing VLANs ----
ip dhcp snooping
ip dhcp snooping vlan 10,20,50,99
no ip dhcp snooping information option
ip arp inspection vlan 10,20,50,99

end
copy running-config startup-config

! ACC1 (SW-ACC1) — access switch #1 (Layer 2 only)
! Po1 = uplink to D1 (Gi0/1-2). Gi0/3 = PC1, Gi0/4 = PC2+Phone, Gi0/5 = AP1, Gi0/6 = PC-ADMIN
! Paste in global config mode (enable -> configure terminal)

hostname SW-ACC1
no ip domain-lookup
ip domain-name lab.local

enable secret Lab-Enable1!
service password-encryption

username admin privilege 15 password Lab-Admin1!

banner motd ^C
***********************************************************
* SW-ACC1 - CCNA-LAB access switch #1.                    *
* Authorized lab use only. All access logged.             *
***********************************************************
^C

crypto key generate rsa general-keys modulus 2048
ip ssh version 2

ip access-list standard MGMT-ACCESS
 permit 192.168.99.0 0.0.0.255
 deny   any

line console 0
 login local
 exec-timeout 10 0
 logging synchronous
line vty 0 15
 transport input ssh
 login local
 access-class MGMT-ACCESS in
 exec-timeout 10 0
 logging synchronous

service timestamps log datetime msec
logging buffered 16384
logging host 192.168.30.10

clock timezone AEST 10
ntp server 192.168.99.2

cdp run
lldp run

! ---- VLANs ----
vlan 10
 name DATA
vlan 20
 name VOICE
vlan 30
 name SERVERS
vlan 50
 name WIRELESS
vlan 99
 name MGMT
vlan 999
 name NATIVE_UNUSED

! ---- Management SVI (L2 switch - no routing, just a mgmt IP) ----
interface Vlan99
 ip address 192.168.99.10 255.255.255.0
 no shutdown
ip default-gateway 192.168.99.1

! ---- Po1: EtherChannel trunk uplink to D1 ----
interface range GigabitEthernet0/1 - 2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate
 channel-group 1 mode active
 no shutdown
interface Port-channel1
 description Uplink to SW-DIST1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,50,99
 switchport nonegotiate
 ip dhcp snooping trust
 ip arp inspection trust

! ---- Access ports ----
interface GigabitEthernet0/3
 description PC1
 switchport mode access
 switchport access vlan 10
 switchport nonegotiate
 spanning-tree portfast edge
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 no shutdown

interface GigabitEthernet0/4
 description PC2 + IP Phone (data + voice)
 switchport mode access
 switchport access vlan 10
 switchport voice vlan 20
 switchport nonegotiate
 spanning-tree portfast edge
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 3
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 no shutdown

interface GigabitEthernet0/5
 description AP1 (lightweight AP, mgmt in VLAN99)
 switchport mode access
 switchport access vlan 99
 switchport nonegotiate
 spanning-tree portfast edge
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 no shutdown

interface GigabitEthernet0/6
 description PC-ADMIN
 switchport mode access
 switchport access vlan 99
 switchport nonegotiate
 spanning-tree portfast edge
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 no shutdown

! ---- Spanning tree ----
spanning-tree mode rapid-pvst
spanning-tree portfast bpduguard default
spanning-tree loopguard default

! ---- Layer 2 threat mitigation ----
ip dhcp snooping
ip dhcp snooping vlan 10,20,50,99
no ip dhcp snooping information option
ip arp inspection vlan 10,20,50,99

end
copy running-config startup-config

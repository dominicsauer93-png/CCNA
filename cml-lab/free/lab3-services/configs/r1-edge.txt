! R1-EDGE - Lab 3 (services). CML: IOSv
! Gi0/1 = SW-DIST1. Lo1 = simulated Internet host 8.8.8.8 (no ISP0 in this lab)
hostname R1-EDGE
no ip domain-lookup
ip domain-name lab.local
!
enable secret Lab-Enable1!
service password-encryption
!
username admin privilege 15 secret Lab-Admin1!
! Account used by CML's console automation (cml-mcp / pyATS). Lab use only.
username cisco privilege 15 secret cisco
!
banner motd #
***********************************************************
* R1-EDGE - CCNA-LAB Internet edge / NAT router.          *
* Authorized lab use only. All access logged.             *
***********************************************************
#
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
!
ip access-list standard MGMT-ACCESS
 permit 192.168.99.0 0.0.0.255
 deny   any
!
line con 0
 login local
 exec-timeout 30 0
 logging synchronous
line vty 0 4
 transport input ssh
 login local
 access-class MGMT-ACCESS in
 exec-timeout 10 0
 logging synchronous
!
service timestamps log datetime msec
logging buffered 16384
logging host 192.168.30.10
!
clock timezone AEST 10 0
!
snmp-server group SNMP-ADMINS v3 priv
snmp-server user snmpadmin SNMP-ADMINS v3 auth sha Lab-SnmpAuth1! priv aes 128 Lab-SnmpPriv1!
snmp-server host 192.168.30.10 version 3 priv snmpadmin
!
interface Loopback0
 description OSPF Router ID
 ip address 1.1.1.1 255.255.255.255
!
interface GigabitEthernet0/1
 description Inside - link to SW-DIST1
 ip address 10.0.12.1 255.255.255.252
 ip ospf network point-to-point
 no shutdown
!
router ospf 1
 router-id 1.1.1.1
 network 1.1.1.1 0.0.0.0 area 0
 network 10.0.12.0 0.0.0.3 area 0
 default-information originate always
!
interface Loopback1
 description Simulated Internet host - ping target
 ip address 8.8.8.8 255.255.255.255
!
! NTP source for this lab (ISP0 is in Lab 2)
ntp master 2
!
interface GigabitEthernet0/0
 description Unused in this lab
 shutdown
!
interface range GigabitEthernet0/2 - 3
 description Unused in this lab
 shutdown
!
end
